> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Afficher la voie de départ (tableau Gares & Connexions) dans la carte Lovelace et l'annoncer par push (voie connue / changement de voie).

**Architecture:** Un nouveau module `boards.py` interroge le tableau des départs de la gare et renvoie `{(numéro, départ théorique): voie}`. Le coordinator ne l'appelle que lorsqu'un train part dans les 10 min (départ réel) ; il mémorise les voies obtenues et les pose sur `Train.platform`. `alerts.py` diffe la voie comme il diffe déjà les retards et produit les alertes `platform` / `platform_changed`, filtrées par le nouveau réglage `platform_notify`.

**Tech Stack:** Home Assistant custom integration (Python 3.13, aiohttp), pytest + `pytest-homeassistant-custom-component` (`aioclient_mock`, `freezer`), carte JS vanilla testée par `node --test`.

**Spec:** `docs/superpowers/specs/2026-10-09-voie-design.md`

```python
"""Gares & Connexions departure boards (platform / voie)."""

from __future__ import annotations

import asyncio
from datetime import datetime
import re
from typing import Any

import aiohttp

BOARD_URL = "https://www.garesetconnexions.sncf/schedule-table/Departures/00{uic}"
HEADERS = {
    "User-Agent": "Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0 Safari/537.36",
    "Accept": "application/json",
}
_UIC_RE = re.compile(r":(\d{8})$")


class BoardError(Exception):
    """Departure board unavailable (unofficial API, never fatal)."""


def uic_from_stop_area(stop_area_id: str | None) -> str | None:
    """'stop_area:SNCF:87723395' -> '87723395'."""
    match = _UIC_RE.search(stop_area_id or "")
    return match.group(1) if match else None


def parse_board(payload: Any) -> dict[tuple[str, datetime], str]:
    """Map (train number, scheduled departure) to the announced track."""
    if not isinstance(payload, list):
        raise BoardError("unexpected payload")
    result: dict[tuple[str, datetime], str] = {}
    for item in payload:
        try:
            number = str(item["trainNumber"])
            scheduled = datetime.fromisoformat(item["scheduledTime"])
            track = str((item.get("platform") or {}).get("track") or "").strip()
        except (KeyError, TypeError, ValueError, AttributeError):
            continue
        if track and scheduled.tzinfo is not None:
            result[(number, scheduled)] = track
    return result


class BoardClient:
    """Thin async client for the station departure board."""

    def __init__(self, session: aiohttp.ClientSession) -> None:
        self._session = session

    async def get_platforms(self, uic: str) -> dict[tuple[str, datetime], str]:
        """Return the tracks currently announced at this station."""
        try:
            async with self._session.get(
                BOARD_URL.format(uic=uic),
                headers=HEADERS,
                timeout=aiohttp.ClientTimeout(total=10),
            ) as resp:
                if resp.status != 200:
                    raise BoardError(f"HTTP {resp.status}")
                payload = await resp.json(content_type=None)
        except (aiohttp.ClientError, asyncio.TimeoutError, ValueError) as err:
            raise BoardError(str(err) or type(err).__name__) from err
        return parse_board(payload)
```

```python
        """Fetch tracks for trains leaving within PLATFORM_LEAD; keep known ones."""
        ids = {t.id for t in trains}
        self._platforms = {k: v for k, v in self._platforms.items() if k in ids}
        due = [t for t in trains if not t.cancelled and t.departure - PLATFORM_LEAD <= now < t.departure]

                        self._platforms[t.id] = platform
        return [replace(t, platform=self._platforms.get(t.id)) for t in trains]
```

```python
def _status_alert(prev: dict, train: Train, threshold: int) -> str | None:
    """Delay / cancellation transition (mutates prev)."""
    delay = train.delay_minutes
    if train.cancelled:
        if prev["cancelled"]:
            return None
        prev["cancelled"] = True
        return "cancelled"
    if prev["cancelled"]:
        prev["cancelled"] = False
        prev["notified_delay"] = delay if delay >= threshold else 0
        return "restored"
    notified = prev["notified_delay"]
    if notified == 0 and delay >= threshold:
        prev["notified_delay"] = delay
        return "delay"
    if notified > 0 and delay >= threshold and abs(delay - notified) >= DELAY_CHANGE_STEP:
        prev["notified_delay"] = delay
        return "delay_changed"
    if notified > 0 and delay < threshold:
        prev["notified_delay"] = 0
        return "on_time"
    return None


def _platform_alert(prev: dict, train: Train, eligible: bool) -> Alert | None:
    """Track announced / changed (mutates prev)."""
    if train.cancelled or not train.platform:
        return None
    seen = prev.get("platform")
    prev["platform"] = train.platform
    if seen and seen != train.platform:
        prev["platform_notified"] = True
        return Alert("platform_changed", train, previous_platform=seen)
    if eligible and not prev.get("platform_notified"):
        prev["platform_notified"] = True
        return Alert("platform", train)
    return None


def compute_alerts(
    state: dict[str, dict],
    trains: list[Train],
    threshold: int,
    platform_notify: str = DEFAULT_PLATFORM_NOTIFY,
) -> list[Alert]:
    """Diff trains against the notified state (mutated in place)."""
    next_id = next((t.id for t in trains if not t.cancelled), None)
    alerts: list[Alert] = []
    for train in trains:
        prev = state.setdefault(train.id, {"notified_delay": 0, "cancelled": False})
        prev["base_departure"] = train.base_departure.isoformat()
        if kind := _status_alert(prev, train, threshold):
            alerts.append(Alert(kind, train))
        eligible = platform_notify == "all" or (platform_notify == "next" and train.id == next_id)
        if alert := _platform_alert(prev, train, eligible):
            alerts.append(alert)
    return alerts
```
